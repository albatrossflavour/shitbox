    ```python
    """DS18B20 1-Wire temperature collector.

    Reads two DS18B20 probes by sensor ID, mapped to semantic roles
    (exterior, engine_bay) per config. IDs are discovered on the Pi 5
    during phase 11 wave 0 and recorded in HARDWARE_IDS.md.

    Graceful degradation: missing or unready probes log and skip the sample
    without taking the engine down (D-24).
    """
    from dataclasses import dataclass
    from typing import Dict, List, Optional

    try:
        from w1thermsensor import NoSensorFoundError, Sensor, SensorNotReadyError, W1ThermSensor
        _HAS_W1 = True
    except ImportError:
        _HAS_W1 = False

    from shitbox.collectors.base import BaseCollector
    from shitbox.storage.models import Reading
    from shitbox.utils.logging import get_logger

    log = get_logger(__name__)


    class DS18B20Collector(BaseCollector[Reading]):
        """Dual DS18B20 probes: exterior + engine_bay roles."""

        def __init__(self, roles: Dict[str, str], sample_rate_hz: float = 1.0, callback=None):
            super().__init__(name="ds18b20", sample_rate_hz=sample_rate_hz, callback=callback)
            self._roles = roles  # {"exterior": "28-xxxx", "engine_bay": "28-yyyy"}
            self._sensors: Dict[str, Optional[W1ThermSensor]] = {}

        def setup(self) -> None:
            if not _HAS_W1:
                log.warning("ds18b20_lib_missing")
                return
            for role, sensor_id in self._roles.items():
                try:
                    self._sensors[role] = W1ThermSensor(sensor_type=Sensor.DS18B20, sensor_id=sensor_id)
                    log.info("ds18b20_sensor_init", role=role, sensor_id=sensor_id)
                except NoSensorFoundError:
                    log.warning("ds18b20_sensor_not_found", role=role, sensor_id=sensor_id)
                    self._sensors[role] = None

        def collect(self) -> Optional[List[Reading]]:
            readings: List[Reading] = []
            for role, sensor in self._sensors.items():
                if sensor is None:
                    continue
                try:
                    temp_c = sensor.get_temperature()
                    readings.append(Reading(sensor_type="temperature", value=temp_c, metadata={"role": role}))
                except (SensorNotReadyError, OSError, Exception) as e:
                    log.warning("ds18b20_read_error", role=role, error=str(e))
            return readings or None
    ```

    ```python
    """Thin INA226 driver over smbus2.

    Written in-tree because there is no maintained Adafruit CircuitPython
    INA226 package on PyPI (researcher confirmed). Register map from the
    TI INA226 datasheet (SBOS547).
    """
    from typing import Tuple

    REG_CONFIG = 0x00
    REG_SHUNT_VOLTAGE = 0x01
    REG_BUS_VOLTAGE = 0x02
    REG_POWER = 0x03
    REG_CURRENT = 0x04
    REG_CALIBRATION = 0x05

    BUS_LSB_V = 0.00125        # 1.25 mV/bit per datasheet
    SHUNT_LSB_V = 0.0000025    # 2.5 µV/bit


    class INA226:
        def __init__(self, bus, address: int = 0x40, shunt_ohms: float = 0.1, max_expected_amps: float = 3.0):
            self._bus = bus
            self._address = address
            self._shunt_ohms = shunt_ohms
            self._current_lsb = max_expected_amps / 32768.0
            self._configure()

        def _read_u16(self, reg: int) -> int:
            data = self._bus.read_i2c_block_data(self._address, reg, 2)
            return (data[0] << 8) | data[1]

        def _read_s16(self, reg: int) -> int:
            val = self._read_u16(reg)
            return val - 0x10000 if val & 0x8000 else val

        def _write_u16(self, reg: int, value: int) -> None:
            self._bus.write_i2c_block_data(self._address, reg, [(value >> 8) & 0xFF, value & 0xFF])

        def _configure(self) -> None:
            # Default config: 16 averages, 1.1 ms conversion time, continuous mode
            self._write_u16(REG_CONFIG, 0x4527)
            cal = int(0.00512 / (self._current_lsb * self._shunt_ohms))
            self._write_u16(REG_CALIBRATION, cal)

        def read(self) -> Tuple[float, float]:
            """Return (bus_voltage_v, current_a)."""
            bus_v = self._read_u16(REG_BUS_VOLTAGE) * BUS_LSB_V
            current_a = self._read_s16(REG_CURRENT) * self._current_lsb
            return bus_v, current_a
    ```
