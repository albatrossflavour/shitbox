    3. **Helper methods** — add these methods to the dashboard() object:

       ```javascript
       async fetchGpsStatus() {
         try {
           const r = await fetch('/api/logbook/gps');
           const data = await r.json();
           this.gpsHasFix = data.has_fix;
           this.gpsStaleMinutes = data.stale_seconds != null
             ? Math.max(1, Math.round(data.stale_seconds / 60)) : 0;
         } catch { this.gpsHasFix = false; this.gpsStaleMinutes = 0; }
       },
       async openNoteModal() {
         this.errorMsg = '';
         this.noteBody = '';
         this.noteEventId = '';
         await this.fetchGpsStatus();
         this.showNoteModal = true;
       },
       async openFuelModal() {
         this.errorMsg = '';
         this.fuelVolume = '';
         this.fuelCost = '';
         this.fuelOdo = '';
         await this.fetchGpsStatus();
         this.showFuelModal = true;
       },
       async saveNote() {
         if (!this.noteBody.trim()) return;
         this.savingNote = true;
         this.errorMsg = '';
         try {
           const body = { body: this.noteBody };
           if (this.noteEventId) body.event_id = parseInt(this.noteEventId, 10);
           const r = await fetch('/api/notes', {
             method: 'POST',
             headers: {'Content-Type': 'application/json'},
             body: JSON.stringify(body)
           });
           if (!r.ok) throw new Error('save failed');
           this.showNoteModal = false;
           this.flashSuccess('Note saved');
         } catch {
           this.errorMsg = 'Could not save — try again';
         } finally { this.savingNote = false; }
       },
       async saveFuel() {
         const vol = parseFloat(this.fuelVolume);
         if (!(vol > 0)) return;
         this.savingFuel = true;
         this.errorMsg = '';
         try {
           const body = { volume_litres: vol };
           if (this.fuelCost) body.cost_aud = parseFloat(this.fuelCost);
           if (this.fuelOdo) body.odometer_km = parseFloat(this.fuelOdo);
           const r = await fetch('/api/fuel', {
             method: 'POST',
             headers: {'Content-Type': 'application/json'},
             body: JSON.stringify(body)
           });
           if (!r.ok) throw new Error('save failed');
           this.showFuelModal = false;
           this.flashSuccess('Fuel stop logged');
         } catch {
           this.errorMsg = 'Could not save — try again';
         } finally { this.savingFuel = false; }
       },
       flashSuccess(msg) {
         this.successMsg = msg;
         setTimeout(() => { this.successMsg = ''; }, 3000);
       },
       ```
