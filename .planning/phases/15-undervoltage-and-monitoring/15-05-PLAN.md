       ```python
       _SYSTEM_ROWS: List[Tuple[str, str, Tuple[str, ...]]] = [
           ("undervoltage", "Undervoltage", ("UNDERVOLTAGE",)),
           ("thermal_throttle", "Thermal Throttle", ("THERMAL_THROTTLE",)),
           ("capture", "Capture", ("CAPTURE_DOWN", "CAPTURE_FAILURE")),  # first-matching = highest severity
       ]
       _RECOVERY_WINDOW_SECONDS: float = 10.0


       def _system_conditions_payload() -> List[Dict[str, Any]]:
           """Return one entry per system condition row for the Health modal."""
           snap = alerts.snapshot()
           now = time.time()
           out: List[Dict[str, Any]] = []
           for key, label, subtype_priority in _SYSTEM_ROWS:
               active_status = None
               for subtype in subtype_priority:
                   st = snap.get(subtype)
                   if st is not None and st.active:
                       active_status = st
                       break
               if active_status is not None:
                   last_seen = active_status.last_fired_ts or None
                   since_ms = int((now - last_seen) * 1000) if last_seen else None
                   out.append({
                       "key": key,
                       "label": label,
                       "state": "active",
                       "subtype": active_status.subtype,
                       "message": active_status.last_message,
                       "last_seen": last_seen,
                       "since_ms": since_ms,
                   })
                   continue
               # No active subtype — check for recent recovery
               restored_status = None
               for subtype in subtype_priority:
                   st = snap.get(subtype)
                   if st is None or st.active:
                       continue
                   if st.last_fired_ts and (now - st.last_fired_ts) <= _RECOVERY_WINDOW_SECONDS:
                       restored_status = st
                       break
               if restored_status is not None:
                   last_seen = restored_status.last_fired_ts or None
                   since_ms = int((now - last_seen) * 1000) if last_seen else None
                   out.append({
                       "key": key,
                       "label": label,
                       "state": "restored",
                       "subtype": restored_status.subtype,
                       "message": restored_status.last_message,
                       "last_seen": last_seen,
                       "since_ms": since_ms,
                   })
                   continue
               # Clear (nothing observed or observation is stale)
               out.append({
                   "key": key,
                   "label": label,
                   "state": "clear",
                   "subtype": None,
                   "message": None,
                   "last_seen": None,
                   "since_ms": None,
               })
           return out
       ```
