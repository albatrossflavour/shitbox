```js
function formatDriveTime(seconds) {
    var h = Math.floor(seconds / 3600);
    var m = Math.floor((seconds % 3600) / 60);
    return h + 'h ' + m + 'm';
}

function renderNotes(notes) {
    var container = document.getElementById('notes-container');
    if (!container) return;
    if (!notes || notes.length === 0) {
        container.innerHTML = '<p style="text-align:center;padding:3rem;color:#8b949e;">No field notes yet.</p>';
        return;
    }
    var html = '';
    for (var i = 0; i < notes.length; i++) {
        var note = notes[i];
        var date = new Date(note.timestamp_utc);
        var dateStr = date.toLocaleDateString('en-AU', { timeZone: TIMEZONE });
        var timeStr = date.toLocaleTimeString('en-AU', { timeZone: TIMEZONE, hour: '2-digit', minute: '2-digit' });
        var meta = '<span>' + dateStr + ' ' + timeStr + '</span>';
        if (note.lat != null && note.lng != null && !note.gps_stale) {
            meta += ' &middot; <a href="https://www.google.com/maps?q=' + note.lat + ',' + note.lng + '" target="_blank" rel="noopener">View location</a>';
        }
        var footer = '';
        if (note.event_id != null) {
            var eventEl = document.getElementById('event-' + note.event_id);
            if (eventEl) {
                footer = '<div class="note-footer">Linked event: <a href="#videos" onclick="document.querySelector(\'[data-section=videos]\').click();">View linked event</a></div>';
            }
        }
        html += '<div class="note-card">';
        html += '<div class="note-meta">' + meta + '</div>';
        html += '<div class="note-body">' + note.body + '</div>';
        html += footer;
        html += '</div>';
    }
    container.innerHTML = html;
}

function injectNoteBadges(notes) {
    if (!notes) return;
    var pencilSvg = '<svg class="note-badge" viewBox="0 0 16 16" fill="currentColor"><path d="M11.013 1.427a1.75 1.75 0 012.474 0l1.086 1.086a1.75 1.75 0 010 2.474l-8.61 8.61c-.21.21-.47.364-.756.445l-3.251.93a.75.75 0 01-.927-.928l.929-3.25c.081-.286.235-.547.445-.758l8.61-8.61zM11.524 2.2l-8.61 8.61a.25.25 0 00-.064.108l-.558 1.953 1.953-.558a.249.249 0 00.108-.064l8.61-8.61a.25.25 0 000-.354l-1.086-1.086a.25.25 0 00-.353 0z"/></svg>';
    for (var i = 0; i < notes.length; i++) {
        if (notes[i].event_id != null) {
            var card = document.getElementById('event-' + notes[i].event_id);
            if (card && !card.querySelector('.note-badge')) {
                card.insertAdjacentHTML('beforeend', pencilSvg);
            }
        }
    }
}

function renderDrivers(data) {
    var loading = document.getElementById('drivers-loading');
    var content = document.getElementById('drivers-content');
    var error = document.getElementById('drivers-error');
    var tbody = document.getElementById('drivers-tbody');
    if (!loading || !content || !tbody) return;

    if (!data || !data.drivers || data.drivers.length === 0) {
        loading.style.display = 'none';
        error.style.display = 'block';
        error.textContent = 'No driver data recorded yet.';
        return;
    }

    var html = '';
    for (var i = 0; i < data.drivers.length; i++) {
        var d = data.drivers[i];
        var isActive = data.active_driver && d.driver_name === data.active_driver;
        var nameClass = isActive ? ' class="active-driver"' : '';
        var nameSuffix = isActive ? ' \u00b7' : '';
        html += '<tr>';
        html += '<td' + nameClass + '>' + d.driver_name + nameSuffix + '</td>';
        html += '<td>' + formatDriveTime(d.total_seconds) + '</td>';
        html += '<td>' + d.pct.toFixed(1) + '%</td>';
        html += '</tr>';
    }
    tbody.innerHTML = html;
    loading.style.display = 'none';
    content.style.display = 'block';
}
```

**Step 3: Add three parallel fetches**

Add these AFTER the existing `fetch('/captures/events.json', ...)` block. They are parallel,
not chained. Each follows the same pattern as the events fetch:

```js
fetch('/captures/notes.json', { cache: 'no-cache' })
    .then(function(r) {
        if (!r.ok) throw new Error(r.status);
        return r.json();
    })
    .then(function(notes) {
        notesData = notes.slice().reverse();
        renderNotes(notesData);
        injectNoteBadges(notesData);
    })
    .catch(function() {});

fetch('/captures/fuel.json', { cache: 'no-cache' })
    .then(function(r) {
        if (!r.ok) throw new Error(r.status);
        return r.json();
    })
    .then(function(fuel) {
        fuelData = fuel;
    })
    .catch(function() {});

fetch('/captures/driver-stats.json', { cache: 'no-cache' })
    .then(function(r) {
        if (!r.ok) throw new Error(r.status);
        return r.json();
    })
    .then(function(data) {
        driverStatsData = data;
        var el = document.getElementById('status-current-driver');
        if (el) {
            el.textContent = data.active_driver || '\u2014';
        }
        renderDrivers(data);
    })
    .catch(function() {
        var loading = document.getElementById('drivers-loading');
        var error = document.getElementById('drivers-error');
        if (loading) loading.style.display = 'none';
        if (error) {
            error.style.display = 'block';
            error.textContent = 'Driver data unavailable.';
        }
    });
```

**Step 5: Fix note badge injection race condition in events fetch handler**

The events fetch and notes fetch run in parallel. If notes.json resolves before events.json,
`injectNoteBadges` finds no event cards in the DOM and silently does nothing. To fix this,
add a call to `injectNoteBadges` in the existing events.json fetch `.then()` handler, AFTER
`renderEvents(events)` and `renderStatus(events)` have completed:

```js
// Inside the existing events.json fetch .then handler, after renderEvents and renderStatus:
if (notesData) injectNoteBadges(notesData);
```

This means badges are injected whichever fetch completes second. The `injectNoteBadges`
function already guards against duplicate insertion via `!card.querySelector('.note-badge')`,
so calling it twice is safe.
