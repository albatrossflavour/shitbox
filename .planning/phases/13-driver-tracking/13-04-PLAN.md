    **3. Methods (add to the Alpine object, next to existing openNotesModal etc.):**
    ```js
    async openDriverModal() {
      try {
        const r = await fetch('/api/driver/stats');
        if (r.ok) {
          const data = await r.json();
          this.driverStats = data.drivers || [];
          this.driverRoster = data.roster || [];
          this.driverSelection = data.active_driver || "";
        }
      } catch (e) { console.error('driver stats fetch failed', e); }
      this.showDriverModal = true;
    },
    async switchDriver(name) {
      try {
        const r = await fetch('/api/driver', {
          method: 'POST',
          headers: {'Content-Type': 'application/json'},
          body: JSON.stringify({name: name}),
        });
        if (r.ok) {
          const data = await r.json();
          this.activeDriver = data.driver_name;
          // Refresh stats so the modal reflects the new stint
          const s = await fetch('/api/driver/stats');
          if (s.ok) {
            const sd = await s.json();
            this.driverStats = sd.drivers || [];
          }
        } else {
          console.warn('driver switch rejected', r.status);
        }
      } catch (e) { console.error('driver switch failed', e); }
    },
    formatDuration(seconds) {
      if (!seconds || seconds <= 0) return '0m';
      const h = Math.floor(seconds / 3600);
      const m = Math.floor((seconds % 3600) / 60);
      return h > 0 ? `${h}h ${m}m` : `${m}m`;
    },
    ```
